# great_gamer in its history (history/countries/ENG - United Kingdom.txt);
# v2.3.4 never uses it. The engine does not evaluate the trigger of an
# is_triggered_only event when an effect fires it (U-2, 2026-10-12: both
# fired), so the decision chooses which one to fire; the triggers stay as a
# second check for events fired some other way.

	title = "tgcx_loadtest.1.t"
	desc = "tgcx_loadtest.1.d"
	picture = "news"
	is_triggered_only = yes

	trigger = {
		ENG = { has_country_flag = great_gamer }
	}

	option = {
		name = "tgcx_loadtest.1.a"
	}
}
